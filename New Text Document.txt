export type UserRole = 'dosje' | 'inspector' | 'institute';

export interface UserAccount {
  id: string;
  username: string;
  passwordHash: string; // Plaintext for local demo verification
  role: UserRole;
  name: string;
  instituteId?: string; // Links institute users to specific institute profiles
}

export interface InstituteProfile {
  id: string;
  name: string;
  org: string;
  location: string;
  expectedAttendance: number;
}

export interface ActivityLog {
  id: string;
  timestamp: string;
  actorUsername: string;
  actorRole: UserRole;
  action: string;
  details: string;
}

// Storage helper for Audit Logs
export const getAuditLogs = (): ActivityLog[] => {
  try {
    return JSON.parse(localStorage.getItem('nigraniAuditLogs') || '[]');
  } catch {
    return [];
  }
};

export const logActivity = (
  actorUsername: string,
  actorRole: UserRole,
  action: string,
  details: string
) => {
  const logs = getAuditLogs();
  const newEntry: ActivityLog = {
    id: 'LOG-' + Date.now(),
    timestamp: new Date().toLocaleString(),
    actorUsername,
    actorRole,
    action,
    details
  };
  localStorage.setItem('nigraniAuditLogs', JSON.stringify([newEntry, ...logs]));
};
